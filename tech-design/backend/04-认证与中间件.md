# 认证与中间件

## 1. JWT 认证

### Token 结构

```go
type Claims struct {
    UserID      uint64 `json:"user_id"`
    UserGroupID uint64 `json:"user_group_id"`
    Email       string `json:"email"`
    jwt.RegisteredClaims
}
```

### Token 生成

```go
func (s *AuthService) GenerateToken(user *model.User) (string, error) {
    claims := &Claims{
        UserID:      user.ID,
        UserGroupID: user.DefaultGroupID,
        Email:       user.Email,
        RegisteredClaims: jwt.RegisteredClaims{
            ExpiresAt: jwt.NewNumericDate(time.Now().Add(24 * time.Hour)),
            IssuedAt:  jwt.NewNumericDate(time.Now()),
            Subject:   strconv.FormatUint(user.ID, 10),
        },
    }

    token := jwt.NewWithClaims(jwt.SigningMethodHS256, claims)
    return token.SignedString([]byte(s.cfg.JWTSecret))
}
```

### Token 解析与验证

```go
func (s *AuthService) ParseToken(tokenString string) (*Claims, error) {
    token, err := jwt.ParseWithClaims(tokenString, &Claims{}, func(token *jwt.Token) (interface{}, error) {
        return []byte(s.cfg.JWTSecret), nil
    })
    if err != nil {
        return nil, err
    }

    claims, ok := token.Claims.(*Claims)
    if !ok || !token.Valid {
        return nil, fmt.Errorf("无效 Token")
    }

    // 检查 Token 是否在黑名单中
    if s.isBlacklisted(tokenString) {
        return nil, fmt.Errorf("Token 已注销")
    }

    return claims, nil
}
```

### Token 黑名单

- 使用 Redis Set 存储，key: `token:blacklist`
- 注销时将 Token 加入黑名单，TTL 等于 Token 剩余有效期
- 验证时检查 Token 是否在黑名单中

## 2. 认证中间件

```go
func AuthMiddleware(authService *AuthService) gin.HandlerFunc {
    return func(c *gin.Context) {
        // 从 Header 提取 Token
        authHeader := c.GetHeader("Authorization")
        if authHeader == "" {
            c.AbortWithStatusJSON(401, gin.H{"error": "缺少认证信息"})
            return
        }

        // 解析 Bearer Token
        parts := strings.SplitN(authHeader, " ", 2)
        if len(parts) != 2 || parts[0] != "Bearer" {
            c.AbortWithStatusJSON(401, gin.H{"error": "认证格式错误"})
            return
        }

        // 验证 Token
        claims, err := authService.ParseToken(parts[1])
        if err != nil {
            c.AbortWithStatusJSON(401, gin.H{"error": "Token 无效或已过期"})
            return
        }

        // 检查用户是否被封禁
        user, err := userService.GetByID(c, claims.UserID)
        if err != nil || user.Blocked {
            c.AbortWithStatusJSON(403, gin.H{"error": "用户已被封禁"})
            return
        }

        // 将用户信息注入 Context
        c.Set("user_id", claims.UserID)
        c.Set("user_group_id", claims.UserGroupID)
        c.Set("email", claims.Email)

        c.Next()
    }
}
```

## 3. 权限中间件

```go
func RequireRole(role string) gin.HandlerFunc {
    return func(c *gin.Context) {
        userID := c.GetUint64("user_id")
        userGroupID := c.GetUint64("user_group_id")

        // 检查用户在当前组中是否拥有指定角色
        hasRole := membershipService.HasRoleInGroup(c, userID, userGroupID, role)
        if !hasRole {
            c.AbortWithStatusJSON(403, gin.H{"error": "权限不足"})
            return
        }

        c.Next()
    }
}
```

## 4. 多租户中间件

```go
func TenantMiddleware() gin.HandlerFunc {
    return func(c *gin.Context) {
        userGroupID := c.GetUint64("user_group_id")

        // 支持通过 Header 切换用户组
        if groupHeader := c.GetHeader("X-User-Group-ID"); groupHeader != "" {
            targetGroupID, _ := strconv.ParseUint(groupHeader, 10, 64)

            // 验证用户是否属于该组
            isMember := membershipService.IsMember(c, c.GetUint64("user_id"), targetGroupID)
            if isMember {
                userGroupID = targetGroupID
                c.Set("user_group_id", userGroupID)
            }
        }

        c.Next()
    }
}
```

### 租户数据隔离

所有 Repository 查询自动注入 `user_group_id` 条件：

```go
func (r *AccountRepo) List(ctx context.Context, userGroupID uint64, params *ListParams) ([]model.Account, int64, error) {
    query := r.db.WithContext(ctx).Model(&model.Account{}).Where("user_group_id = ?", userGroupID)

    // 应用过滤条件
    if params.Type != "" {
        query = query.Where("account_type = ?", params.Type)
    }
    if params.Active != nil {
        query = query.Where("is_active = ?", *params.Active)
    }

    // 分页
    var total int64
    query.Count(&total)

    var accounts []model.Account
    query.Offset((params.Page - 1) * params.PerPage).Limit(params.PerPage).Find(&accounts)

    return accounts, total, nil
}
```

## 6. 请求限流中间件

```go
func RateLimitMiddleware(redisClient *redis.Client) gin.HandlerFunc {
    limiter := rate.NewLimiter(rate.Every(time.Second), 60) // 每秒 60 个请求

    return func(c *gin.Context) {
        userID := c.GetUint64("user_id")
        key := fmt.Sprintf("ratelimit:%d", userID)

        // 使用 Redis 滑动窗口限流
        count, _ := redisClient.Incr(c, key).Result()
        if count == 1 {
            redisClient.Expire(c, key, time.Minute)
        }

        if count > 60 {
            c.AbortWithStatusJSON(429, gin.H{"error": "请求过于频繁"})
            return
        }

        c.Next()
    }
}
```

## 7. 请求日志中间件

```go
func LoggerMiddleware(logger *zap.Logger) gin.HandlerFunc {
    return func(c *gin.Context) {
        start := time.Now()
        path := c.Request.URL.Path

        c.Next()

        latency := time.Since(start)
        status := c.Writer.Status()
        method := c.Request.Method
        userID, _ := c.Get("user_id")

        logger.Info("HTTP Request",
            zap.String("method", method),
            zap.String("path", path),
            zap.Int("status", status),
            zap.Duration("latency", latency),
            zap.Any("user_id", userID),
            zap.String("client_ip", c.ClientIP()),
        )
    }
}
```

## 8. 双因素认证 (2FA)

### 启用流程

```
1. 用户请求启用 2FA
2. 生成 TOTP Secret（基于 github.com/pquerna/otp/totp）
3. 返回 QR Code URL（otpauth://totp/...）
4. 用户扫描 QR Code 并输入验证码
5. 验证通过后保存 Secret，标记 2FA 已启用
```

### 登录流程

```
1. 用户提交 email + password
2. 验证密码正确
3. 检查 2FA 是否启用
4. 如果启用：
   a. 返回临时 Token（仅用于 2FA 验证）
   b. 前端显示 2FA 输入框
   c. 用户输入 TOTP 验证码
   d. 验证通过后返回完整 Token
5. 如果未启用：直接返回完整 Token
```

```go
func SetupRouter(
    cfg *config.Config,
    authHandler *handler.AuthHandler,
    accountHandler *handler.AccountHandler,
    transactionHandler *handler.TransactionHandler,
    // ... 其他 handler
) *gin.Engine {
    r := gin.New()

    // 全局中间件
    r.Use(middleware.LoggerMiddleware(logger))
    r.Use(middleware.CORSMiddleware())
    r.Use(middleware.RecoveryMiddleware())
    r.Use(middleware.RateLimitMiddleware(redisClient))

    // 健康检查
    r.GET("/health", func(c *gin.Context) {
        c.JSON(200, gin.H{"status": "ok"})
    })

    // API v1
    v1 := r.Group("/api/v1")
    {
        // 公开接口（无需认证）
        auth := v1.Group("/auth")
        {
            auth.POST("/register", authHandler.Register)
            auth.POST("/login", authHandler.Login)
            auth.POST("/refresh", authHandler.Refresh)
            auth.POST("/2fa/verify", authHandler.Verify2FA)
        }

        // 需要认证的接口
        authenticated := v1.Group("")
        authenticated.Use(middleware.AuthMiddleware(authService))
        authenticated.Use(middleware.TenantMiddleware())
        {
            // 认证相关
            authenticated.POST("/auth/logout", authHandler.Logout)
            authenticated.POST("/auth/2fa/enable", authHandler.Enable2FA)

            // 账户
            accounts := authenticated.Group("/accounts")
            {
                accounts.GET("", accountHandler.List)
                accounts.POST("", accountHandler.Create)
                accounts.GET("/:id", accountHandler.Get)
                accounts.PUT("/:id", accountHandler.Update)
                accounts.DELETE("/:id", accountHandler.Delete)
                accounts.GET("/:id/transactions", accountHandler.ListTransactions)
                accounts.GET("/:id/piggy-banks", accountHandler.ListPiggyBanks)
                accounts.GET("/:id/attachments", accountHandler.ListAttachments)
            }

            // 交易
            transactions := authenticated.Group("/transactions")
            {
                transactions.GET("", transactionHandler.List)
                transactions.POST("", transactionHandler.Create)
                transactions.GET("/:id", transactionHandler.Get)
                transactions.PUT("/:id", transactionHandler.Update)

                transactions.GET("/:id/attachments", transactionHandler.ListAttachments)
                transactions.GET("/:id/piggy-bank-events", transactionHandler.ListPiggyBankEvents)
            }

            // ... 其他资源路由
        }
    }

    return r
}
```
