# 核心业务逻辑

## 1. 交易创建流程

### 流程图

```
客户端 POST /transactions
    │
    ▼
Handler: 参数绑定与校验
    │
    ▼
Service: CreateTransaction
    │
    ├── 1. 校验交易类型与账户匹配
    │      ├── Withdrawal: source ∈ Asset, destination ∈ Expense
    │      ├── Deposit: source ∈ Revenue, destination ∈ Asset
    │      └── Transfer: source ∈ Asset, destination ∈ Asset (source ≠ destination)
    │
    ├── 2. 创建 TransactionGroup
    │
    ├── 3. 遍历 transactions 数组，创建 TransactionJournal
    │      ├── 设置日期、描述、类型
    │      ├── 关联分类、预算、账单
    │      └── 创建两条 Transaction（借方 + 贷方）
    │           ├── 借方: amount < 0, account = source
    │           └── 贷方: amount > 0, account = destination
    │
    ├── 4. 处理多币种
    │      ├── 计算 native_amount（本币等值）
    │      └── 计算 native_foreign_amount（外币本币等值）
    │
    ├── 5. 账单自动匹配
    │      └── 检查交易描述/金额/日期是否匹配账单规则
    │
    ├── 6. 异步触发规则引擎
    │      └── goroutine: ExecuteRulesForTransaction(journal)
    │
    ├── 7. 异步触发 Webhook
    │      └── goroutine: TriggerWebhooks(STORE_TRANSACTION, journal)
    │
    └── 8. 返回创建结果
```

### 关键实现

```go
func (s *TransactionService) Create(ctx context.Context, req *dto.CreateTransactionReq) (*dto.TransactionGroupResp, error) {
    // 1. 开启数据库事务
    tx := s.db.Begin()
    defer func() {
        if r := recover(); r != nil {
            tx.Rollback()
        }
    }()

    // 2. 校验交易类型与账户
    if err := s.validateTransactionAccounts(req); err != nil {
        return nil, err
    }

    // 3. 创建交易组
    group := &model.TransactionGroup{
        UserID:      req.UserID,
        UserGroupID: req.UserGroupID,
        Title:       req.Title,
    }
    if err := tx.Create(group).Error; err != nil {
        tx.Rollback()
        return nil, err
    }

    // 4. 创建交易日志和交易记录
    for i, trReq := range req.Transactions {
        journal, err := s.createJournal(tx, group.ID, req, &trReq, i)
        if err != nil {
            tx.Rollback()
            return nil, err
        }

        // 创建借方和贷方交易记录
        if err := s.createTransactionPair(tx, journal, &trReq); err != nil {
            tx.Rollback()
            return nil, err
        }
    }

    // 5. 提交事务
    if err := tx.Commit().Error; err != nil {
        return nil, err
    }

    // 6. 异步触发规则引擎和 Webhook
    go s.ruleEngine.ExecuteForTransaction(ctx, group)
    go s.webhookService.Trigger(ctx, "STORE_TRANSACTION", group)

    return s.toResponse(group), nil
}
```

## 2. 账户余额计算

### 余额计算逻辑

```go
func (s *AccountService) GetBalance(ctx context.Context, accountID uint64, date time.Time) (decimal.Decimal, error) {
    account, _ := s.repo.GetByID(ctx, accountID)

    // 1. 获取期初余额（Opening Balance）
    openingBalance := s.getOpeningBalance(ctx, accountID)

    // 2. 计算截至指定日期的所有交易净额
    var sum decimal.Decimal
    switch account.AccountType {
    case "asset", "default", "cash", "credit_card", "saving", "shared_asset":
        // 资产账户：贷方增加余额，借方减少余额
        sum = s.repo.SumTransactionsUntilDate(ctx, accountID, date)
    case "debt", "loan", "mortgage":
        // 负债账户：借方增加负债，贷方减少负债
        sum = s.repo.SumTransactionsUntilDate(ctx, accountID, date).Neg()
    case "expense", "revenue":
        // 收支账户不计算余额
        return decimal.Zero, nil
    }

    return openingBalance.Add(sum), nil
}
```

### 余额缓存策略

- 账户余额按日期缓存到 Redis，key: `account:{id}:balance:{date}`
- 缓存 TTL: 5 分钟
- 交易创建/更新/删除时清除相关账户的余额缓存

## 3. 规则引擎

### 执行流程

```go
func (e *RuleEngine) ExecuteForTransaction(ctx context.Context, group *model.TransactionGroup) {
    // 1. 获取所有活跃规则组（按 order 排序）
    groups, _ := e.ruleGroupRepo.ListActive(ctx, group.UserGroupID)

    for _, rg := range groups {
        // 2. 获取组内活跃规则
        rules, _ := e.ruleRepo.ListActiveByGroup(ctx, rg.ID)

        for _, rule := range rules {
            // 3. 检查触发器
            matched := e.checkTriggers(rule, group)

            if matched {
                // 4. 执行动作
                e.executeActions(ctx, rule, group)
            }

            if rule.StopProcessing {
                break
            }
        }

        if rg.StopProcessing {
            break
        }
    }
}
```

### 触发器检查

```go
func (e *RuleEngine) checkTriggers(rule model.Rule, group *model.TransactionGroup) bool {
    triggers := e.getActiveTriggers(rule)

    if rule.Strict {
        // 严格模式：所有触发器必须匹配（AND）
        for _, t := range triggers {
            if !e.matchTrigger(t, group) {
                return false
            }
        }
        return true
    } else {
        // 非严格模式：任一触发器匹配即可（OR）
        for _, t := range triggers {
            if e.matchTrigger(t, group) {
                return true
            }
        }
        return false
    }
}
```

### 触发器匹配实现

```go
func (e *RuleEngine) matchTrigger(trigger model.RuleTrigger, group *model.TransactionGroup) bool {
    switch trigger.TriggerType {
    case "description_contains":
        return strings.Contains(group.Journals[0].Description, trigger.TriggerValue)
    case "description_starts":
        return strings.HasPrefix(group.Journals[0].Description, trigger.TriggerValue)
    case "amount_less":
        amount, _ := decimal.NewFromString(trigger.TriggerValue)
        return group.Journals[0].Transactions[0].Amount.LessThan(amount.Abs())
    case "amount_more":
        amount, _ := decimal.NewFromString(trigger.TriggerValue)
        return group.Journals[0].Transactions[0].Amount.GreaterThan(amount.Abs())
    case "source_is":
        return group.Journals[0].Transactions[0].Account.Name == trigger.TriggerValue
    case "category_is":
        if group.Journals[0].Category != nil {
            return group.Journals[0].Category.Name == trigger.TriggerValue
        }
        return false
    case "has_tag":
        for _, tag := range group.Journals[0].Tags {
            if tag.Tag == trigger.TriggerValue {
                return true
            }
        }
        return false
    // ... 其他触发器类型
    default:
        return false
    }
}
```

```go
func (e *RuleEngine) executeActions(ctx context.Context, rule model.Rule, group *model.TransactionGroup) {
    actions := e.getActiveActions(rule)

    for _, action := range actions {
        // 解析表达式语言
        value := e.resolveExpression(action.ActionValue, group)

        switch action.ActionType {
        case "set_category":
            e.setCategory(ctx, group, value)
        case "add_tag":
            e.addTag(ctx, group, value)
        case "set_description":
            e.setDescription(ctx, group, value)
        case "set_amount":
            e.setAmount(ctx, group, value)
        case "link_to_bill":
            e.linkToBill(ctx, group, value)
        case "convert_withdrawal":
            e.convertTransactionType(ctx, group, "withdrawal")
        // ... 其他动作类型
        }

        if action.StopProcessing {
            break
        }
    }
}
```

### 表达式引擎

```go
// resolveExpression 解析动作值中的表达式
// 如果值以 "=" 开头，则作为表达式计算
// 否则作为字面值返回
func (e *RuleEngine) resolveExpression(value string, group *model.TransactionGroup) string {
    if !strings.HasPrefix(value, "=") {
        return value
    }

    expr := value[1:]

    // 构建表达式上下文变量
    vars := map[string]interface{}{
        "amount":             group.Journals[0].Transactions[0].Amount.String(),
        "description":        group.Journals[0].Description,
        "date":               group.Journals[0].Date.Format("2006-01-02"),
        "source_account":     getSourceAccountName(group),
        "destination_account": getDestinationAccountName(group),
        "category":           getCategoryName(group),
        "budget":             getBudgetName(group),
        "bill":               getBillName(group),
        "tags":               getTagNames(group),
    }

    // 使用简单的表达式求值器
    result, err := e.exprEngine.Evaluate(expr, vars)
    if err != nil {
        return value // 表达式错误时返回原始值
    }

    return result
}
```

```go
func (s *CurrencyService) Convert(ctx context.Context, fromCurrencyID, toCurrencyID uint64, amount decimal.Decimal, date time.Time) (decimal.Decimal, error) {
    // 同币种直接返回
    if fromCurrencyID == toCurrencyID {
        return amount, nil
    }

    // 查询汇率（优先精确日期，其次最近历史）
    rate, err := s.getRate(ctx, fromCurrencyID, toCurrencyID, date)
    if err != nil {
        return decimal.Zero, fmt.Errorf("汇率未找到: %d -> %d", fromCurrencyID, toCurrencyID)
    }

    return amount.Mul(rate), nil
}

func (s *CurrencyService) getRate(ctx context.Context, fromID, toID uint64, date time.Time) (decimal.Decimal, error) {
    // 1. 尝试精确日期
    rate, err := s.repo.GetRate(ctx, fromID, toID, date)
    if err == nil {
        return rate.Rate, nil
    }

    // 2. 查询最近的历史汇率
    rate, err = s.repo.GetLatestRateBefore(ctx, fromID, toID, date)
    if err == nil {
        return rate.Rate, nil
    }

    // 3. 尝试反向汇率
    reverseRate, err := s.repo.GetRate(ctx, toID, fromID, date)
    if err == nil {
        return decimal.NewFromInt(1).Div(reverseRate.Rate), nil
    }

    return decimal.Zero, fmt.Errorf("无可用汇率")
}
```

### Native Amount 计算

```go
func (s *TransactionService) calculateNativeAmounts(ctx context.Context, journal *model.TransactionJournal, transactions *[]model.Transaction) error {
    defaultCurrency := s.getDefaultCurrency(ctx, journal.UserGroupID)

    for i := range *transactions {
        tr := &(*transactions)[i]

        // native_amount: 交易币种金额转换为本币
        if tr.CurrencyID == defaultCurrency.ID {
            tr.NativeAmount = tr.Amount
        } else {
            native, err := s.currencyService.Convert(ctx, tr.CurrencyID, defaultCurrency.ID, tr.Amount, journal.Date)
            if err != nil {
                tr.NativeAmount = tr.Amount // 降级：使用原始金额
            } else {
                tr.NativeAmount = native
            }
        }

        // native_foreign_amount: 外币金额转换为本币
        if tr.ForeignCurrencyID != nil && tr.ForeignAmount != nil {
            if *tr.ForeignCurrencyID == defaultCurrency.ID {
                tr.NativeForeignAmount = tr.ForeignAmount
            } else {
                native, err := s.currencyService.Convert(ctx, *tr.ForeignCurrencyID, defaultCurrency.ID, *tr.ForeignAmount, journal.Date)
                if err != nil {
                    tr.NativeForeignAmount = tr.ForeignAmount
                } else {
                    tr.NativeForeignAmount = &native
                }
            }
        }
    }

    return nil
}
```

```go
func (s *RecurrenceService) ProcessDueRecurrences(ctx context.Context) {
    today := time.Now().Truncate(24 * time.Hour)

    // 获取所有到期且活跃的定期交易
    recurrences, _ := s.repo.ListDueAndActive(ctx, today)

    for _, rec := range recurrences {
        // 检查重复条件
        if !s.shouldFire(rec, today) {
            continue
        }

        // 根据模板创建交易
        group, err := s.createTransactionFromRecurrence(ctx, &rec, today)
        if err != nil {
            s.logger.Error("定期交易创建失败", zap.Uint64("recurrence_id", rec.ID), zap.Error(err))
            continue
        }

        // 更新最近执行日期
        s.repo.UpdateLatestDate(ctx, rec.ID, today)

        // 如果设置了 apply_rules，触发规则引擎
        if rec.ApplyRules {
            go s.ruleEngine.ExecuteForTransaction(ctx, group)
        }
    }
}
```

```go
func (s *WebhookService) Trigger(ctx context.Context, event string, data interface{}) {
    // 获取匹配触发条件的活跃 Webhook
    webhooks, _ := s.repo.ListActiveByTrigger(ctx, event)

    for _, wh := range webhooks {
        // 构建消息体
        message := s.buildMessage(wh, event, data)

        // 创建消息记录
        msg := &model.WebhookMessage{
            WebhookID: wh.ID,
            UUID:      uuid.New().String(),
            Message:   message,
        }
        s.repo.CreateMessage(ctx, msg)

        // 异步发送
        go s.send(ctx, wh, msg)
    }
}

func (s *WebhookService) send(ctx context.Context, webhook model.Webhook, msg *model.WebhookMessage) {
    // HMAC 签名
    signature := s.sign(webhook.Secret, msg.Message)

    // 发送 HTTP POST
    req, _ := http.NewRequest("POST", webhook.URL, strings.NewReader(msg.Message))
    req.Header.Set("Content-Type", "application/json")
    req.Header.Set("X-Webhook-Signature", signature)

    resp, err := http.DefaultClient.Do(req)

    // 记录尝试
    attempt := &model.WebhookAttempt{
        WebhookMessageID: msg.ID,
    }
    if err != nil {
        attempt.Status = 0
        attempt.Logs = err.Error()
        msg.Errored = true
    } else {
        defer resp.Body.Close()
        attempt.Status = resp.StatusCode
        if resp.StatusCode >= 200 && resp.StatusCode < 300 {
            msg.Sent = true
        } else {
            msg.Errored = true
        }
    }

    s.repo.CreateAttempt(ctx, attempt)
    s.repo.UpdateMessage(ctx, msg)
}
```

```go
func (s *SearchService) SearchTransactions(ctx context.Context, query string, page, perPage int) ([]model.TransactionGroup, int64, error) {
    operators := ParseSearchQuery(query)

    // 构建 GORM 查询
    db := s.db.Model(&model.TransactionGroup{})

    for _, op := range operators {
        switch op.Type {
        case "description_contains":
            db = db.Joins("JOIN transaction_journals ON transaction_journals.group_id = transaction_groups.id").
                Where("transaction_journals.description LIKE ?", "%"+op.Value+"%")
        case "amount_min":
            db = db.Joins("JOIN transaction_journals tj ON tj.group_id = transaction_groups.id").
                Joins("JOIN transactions t ON t.journal_id = tj.id").
                Where("ABS(t.amount) >= ?", op.Value)
        case "date_after":
            db = db.Joins("JOIN transaction_journals ON transaction_journals.group_id = transaction_groups.id").
                Where("transaction_journals.date >= ?", op.Value)
        case "category_is":
            db = db.Joins("JOIN transaction_journals ON transaction_journals.group_id = transaction_groups.id").
                Joins("JOIN categories ON categories.id = transaction_journals.category_id").
                Where("categories.name = ?", op.Value)
        // ... 其他操作符
        }
    }

    // 分页
    var total int64
    db.Count(&total)

    var groups []model.TransactionGroup
    db.Offset((page - 1) * perPage).Limit(perPage).Find(&groups)

    return groups, total, nil
}
```

```go
func (s *BillService) AutoMatch(ctx context.Context, journal *model.TransactionJournal) {
    // 获取所有活跃账单
    bills, _ := s.repo.ListActive(ctx, journal.UserGroupID)

    for _, bill := range bills {
        if !bill.Automatch {
            continue
        }

        // 检查匹配规则
        if s.matchBill(bill, journal) {
            // 关联账单
            s.repo.AssociateJournal(ctx, bill.ID, journal.ID)
            break // 一个交易只关联一个账单
        }
    }
}

func (s *BillService) CheckBillReminders(ctx context.Context) {
    // 账单提醒提前天数：90、30、14、7、0 天
    reminderDays := []int{90, 30, 14, 7, 0}
    bills, _ := s.repo.ListActive(ctx, 0) // 获取所有活跃账单

    for _, bill := range bills {
        for _, days := range reminderDays {
            reminderDate := bill.Date.AddDate(0, 0, -days)
            if time.Now().After(reminderDate) && !s.hasReminderSent(ctx, bill.ID, days) {
                s.notificationService.Send(ctx, &BillReminderNotification{BillID: bill.ID, DaysBefore: days})
                s.markReminderSent(ctx, bill.ID, days)
            }
        }
    }
}

func (s *BillService) matchBill(bill model.Bill, journal *model.TransactionJournal) bool {
    // 1. 检查描述匹配
```

```go
    for _, row := range rows {

            continue
        }

        // 4. 创建交易
        s.transactionService.Create(ctx, transaction)
    }

    // 5. 返回导入结果
    return &ImportResult{Imported: imported, Skipped: skipped, Errors: errors}, nil
}
```

```go
func (s *ImportService) isDuplicate(ctx context.Context, transaction *dto.CreateTransactionReq) bool {
    // 基于 import_hash 检测重复
    hash := s.computeImportHash(transaction)
    exists, _ := s.journalMetaRepo.ExistsByNameAndData(ctx, "import_hash", hash)
    return exists
}

func (s *ImportService) computeImportHash(transaction *dto.CreateTransactionReq) string {
    // 组合：日期 + 描述 + 金额 + 来源账户 + 目标账户
    raw := fmt.Sprintf("%s|%s|%s|%d|%d",
        transaction.Date,
        transaction.Description,
        transaction.Transactions[0].Amount,
        transaction.Transactions[0].SourceID,
        transaction.Transactions[0].DestinationID,
    )
    return fmt.Sprintf("%x", md5.Sum([]byte(raw)))
}
```

```go
    // 1. 确定收件地址（4级优先级）
    //    a. 用户自定义通知路由方法
    //    b. 用户偏好 remote_guard_alt_email（备用邮箱）
    //    c. demo 角色用户发送给站点所有者
    //    d. 默认使用用户注册邮箱

    msg := s.buildMessage(s.cfg.FromAddress, to, notification)

    addr := fmt.Sprintf("%s:%d", s.cfg.Host, s.cfg.Port)
    return smtp.SendMail(addr, auth, s.cfg.FromAddress, []string{to}, msg)
}
```

```go
    if err != nil {
        return err
    }
    defer resp.Body.Close()
    return nil
}
```

```go
    // 1. 获取用户 Token
    userToken := s.getUserToken(ctx, user)

    // 2. 构建 Pushover 请求
    payload := url.Values{}
    payload.Set("token", s.cfg.AppToken)
    payload.Set("user", userToken)
    payload.Set("message", notification.Message())

    // 3. 发送 HTTP POST
    resp, err := s.httpclient.PostForm("https://api.pushover.net/1/messages.json", payload)
    if err != nil {
        return err
    }
    defer resp.Body.Close()
    return nil
}
```
