# 核心组件设计

## 1. 交易表单组件 (TransactionForm)

这是系统最核心的组件，需要处理单笔交易、拆分交易、多币种等复杂场景。

### 组件设计

```vue
<!-- components/transaction/TransactionForm.vue -->
<script setup lang="ts">
import { ref, computed, watch } from 'vue'
import { useI18n } from 'vue-i18n'
import { transactionApi } from '@/api/transactions'
import { useCurrencyStore } from '@/stores/currency'
import AmountInput from '@/components/form/AmountInput.vue'
import AccountSelect from '@/components/form/AccountSelect.vue'
import CategorySelect from '@/components/form/CategorySelect.vue'
import TagSelect from '@/components/form/TagSelect.vue'
import BudgetSelect from '@/components/form/BudgetSelect.vue'
import CurrencySelect from '@/components/form/CurrencySelect.vue'
import TransactionSplit from './TransactionSplit.vue'

const props = defineProps<{
  initialData?: TransactionGroup
}>()

const emit = defineEmits<{
  saved: [group: TransactionGroup]
  cancel: []
}>()

const { t } = useI18n()
const currencyStore = useCurrencyStore()

// 交易类型
const transactionType = ref<TransactionType>(props.initialData?.journals[0].transaction_type_id || 'withdrawal')

// 基础信息
const description = ref(props.initialData?.journals[0].description || '')
const date = ref(props.initialData?.journals[0].date || dayjs().format('YYYY-MM-DD'))

// 拆分交易
const splits = ref<TransactionItemForm[]>([])
const hasSplit = ref(false)

// 初始化
if (props.initialData) {
  splits.value = props.initialData.journals.map(j => ({
    source_id: j.transactions.find(t => new Decimal(t.amount).lt(0))?.account_id || 0,
    destination_id: j.transactions.find(t => new Decimal(t.amount).gt(0))?.account_id || 0,
    amount: j.transactions.find(t => new Decimal(t.amount).gt(0))?.amount || '0',
    currency_id: j.currency_id,
    category_id: j.category_id,
    budget_id: j.budget_id,
    tags: j.tags.map(tag => tag.tag),
    description: j.description,
  }))
} else {
  // 默认一条空拆分
  splits.value = [{
    source_id: 0,
    destination_id: 0,
    amount: '0',
    currency_id: currencyStore.defaultCurrency?.id || 1,
  }]
}

// 来源账户类型（根据交易类型）
const sourceAccountType = computed(() => {
  switch (transactionType.value) {
    case 'withdrawal': return 'asset'
    case 'deposit': return 'revenue'
    case 'transfer': return 'asset'
    default: return 'asset'
  }
})

// 目标账户类型
const destinationAccountType = computed(() => {
  switch (transactionType.value) {
    case 'withdrawal': return 'expense'
    case 'deposit': return 'asset'
    case 'transfer': return 'asset'
    default: return 'expense'
  }
})

// 添加拆分
function addSplit() {
  hasSplit.value = true
  splits.value.push({
    source_id: splits.value[0].source_id,
    destination_id: splits.value[0].destination_id,
    amount: '0',
    currency_id: splits.value[0].currency_id,
    description: '',
  })
}

// 移除拆分
function removeSplit(index: number) {
  splits.value.splice(index, 1)
  if (splits.value.length <= 1) {
    hasSplit.value = false
  }
}

// 保存
const saving = ref(false)
async function save() {
  saving.value = true
  try {
    const form: TransactionForm = {
      type: transactionType.value,
      date: date.value,
      description: description.value,
      title: hasSplit.value ? description.value : undefined,
      transactions: splits.value,
    }

    let result: TransactionGroup
    if (props.initialData) {
      result = (await transactionApi.update(props.initialData.id, form)).data
    } else {
      result = (await transactionApi.create(form)).data
    }
    emit('saved', result)
  } finally {
    saving.value = false
  }
}
</script>

<template>
  <div class="transaction-form">
    <!-- 交易类型选择 -->
    <div class="type-selector">
      <el-radio-group v-model="transactionType" size="large">
        <el-radio-button value="withdrawal">{{ t('transaction.withdrawal') }}</el-radio-button>
        <el-radio-button value="deposit">{{ t('transaction.deposit') }}</el-radio-button>
        <el-radio-button value="transfer">{{ t('transaction.transfer') }}</el-radio-button>
      </el-radio-group>
    </div>

    <!-- 基础信息 -->
    <el-form label-position="top">
      <el-form-item :label="t('transaction.description')">
        <el-input v-model="description" :placeholder="t('transaction.descriptionPlaceholder')" />
      </el-form-item>

      <el-form-item :label="t('transaction.date')">
        <el-date-picker v-model="date" type="date" value-format="YYYY-MM-DD" />
      </el-form-item>

      <!-- 拆分交易列表 -->
      <div v-for="(split, index) in splits" :key="index" class="split-item">
        <div v-if="hasSplit" class="split-header">
          <span>{{ t('transaction.splitN', { n: index + 1 }) }}</span>
          <el-button text type="danger" @click="removeSplit(index)">{{ t('common.remove') }}</el-button>
        </div>

        <div class="split-form" :class="{ 'is-split': hasSplit }">
          <!-- 来源账户 -->
          <el-form-item :label="t('transaction.sourceAccount')">
            <AccountSelect v-model="split.source_id" :type="sourceAccountType" />
          </el-form-item>

          <!-- 目标账户 -->
          <el-form-item :label="t('transaction.destinationAccount')">
            <AccountSelect v-model="split.destination_id" :type="destinationAccountType" />
          </el-form-item>

          <!-- 金额 -->
          <el-form-item :label="t('transaction.amount')">
            <div class="amount-row">
              <AmountInput v-model="split.amount" />
              <CurrencySelect v-model="split.currency_id" style="width: 120px" />
            </div>
          </el-form-item>

          <!-- 外币金额（可选） -->
          <el-form-item :label="t('transaction.foreignAmount')">
            <div class="amount-row">
              <AmountInput v-model="split.foreign_amount" :optional="true" />
              <CurrencySelect v-model="split.foreign_currency_id" :optional="true" style="width: 120px" />
            </div>
          </el-form-item>

          <!-- 分类 -->
          <el-form-item :label="t('transaction.category')">
            <CategorySelect v-model="split.category_id" />
          </el-form-item>

          <!-- 预算 -->
          <el-form-item v-if="transactionType === 'withdrawal'" :label="t('transaction.budget')">
            <BudgetSelect v-model="split.budget_id" />
          </el-form-item>

          <!-- 标签 -->
          <el-form-item :label="t('transaction.tags')">
            <TagSelect v-model="split.tags" />
          </el-form-item>

          <!-- 拆分描述 -->
          <el-form-item v-if="hasSplit" :label="t('transaction.splitDescription')">
            <el-input v-model="split.description" />
          </el-form-item>
        </div>
      </div>

      <!-- 添加拆分按钮 -->
      <el-button type="primary" text @click="addSplit">
        <el-icon><Plus /></el-icon> {{ t('transaction.addSplit') }}
      </el-button>

      <!-- 操作按钮 -->
      <div class="form-actions">
        <el-button @click="emit('cancel')">{{ t('common.cancel') }}</el-button>
        <el-button type="primary" :loading="saving" @click="save">{{ t('common.save') }}</el-button>
      </div>
    </el-form>
  </div>
</template>
```

## 2. 金额输入组件 (AmountInput)

```vue
<!-- components/form/AmountInput.vue -->
<script setup lang="ts">
import { ref, watch, computed } from 'vue'
import Decimal from 'decimal.js'

const props = defineProps<{
  modelValue?: string
  optional?: boolean
  disabled?: boolean
}>()

const emit = defineEmits<{
  'update:modelValue': [value: string | undefined]
}>()

const displayValue = ref('')

// 格式化显示
watch(() => props.modelValue, (val) => {
  if (val) {
    const d = new Decimal(val)
    displayValue.value = d.toFixed(2)
  } else {
    displayValue.value = ''
  }
}, { immediate: true })

function handleInput(val: string) {
  // 只允许数字、小数点、负号
  const cleaned = val.replace(/[^\d.\-]/g, '')
  displayValue.value = cleaned
}

function handleBlur() {
  if (displayValue.value) {
    try {
      const d = new Decimal(displayValue.value)
      emit('update:modelValue', d.toFixed(2))
    } catch {
      emit('update:modelValue', '0.00')
    }
  } else {
    emit('update:modelValue', undefined)
  }
}
</script>

<template>
  <el-input
    v-model="displayValue"
    :disabled="disabled"
    @input="handleInput"
    @blur="handleBlur"
    placeholder="0.00"
    clearable
  >
    <template #prefix>
      <span class="text-gray-400">¥</span>
    </template>
  </el-input>
</template>
```

## 3. 账户选择组件 (AccountSelect)

```vue
<!-- components/form/AccountSelect.vue -->
<script setup lang="ts">
import { ref, onMounted } from 'vue'
import { accountApi } from '@/api/accounts'
import { useAutocomplete } from '@/composables/useAutocomplete'

const props = defineProps<{
  modelValue?: number
  type?: string
  clearable?: boolean
}>()

const emit = defineEmits<{
  'update:modelValue': [value: number | undefined]
}>()

const { items, loading, search } = useAutocomplete('accounts')
const selectedId = ref(props.modelValue)

// 初始加载
onMounted(async () => {
  if (props.type) {
    const { data } = await accountApi.list({ type: props.type, per_page: 500 })
    items.value = data.map(a => ({ id: a.id, name: a.name }))
  }
})

function handleQuery(query: string) {
  search(query)
}

function handleChange(id: number) {
  emit('update:modelValue', id)
}
</script>

<template>
  <el-select
    v-model="selectedId"
    filterable
    remote
    :remote-method="handleQuery"
    :loading="loading"
    :clearable="clearable"
    :placeholder="t('form.selectAccount')"
    @change="handleChange"
  >
    <el-option
      v-for="item in items"
      :key="item.id"
      :label="item.name"
      :value="item.id"
    />
  </el-select>
</template>
```

## 4. 图表组件

### 余额趋势图 (BalanceTrendChart)

```vue
<!-- components/chart/BalanceTrendChart.vue -->
<script setup lang="ts">
import { ref, onMounted, watch } from 'vue'
import * as echarts from 'echarts'
import { chartApi } from '@/api/charts'
import { useDateRange } from '@/composables/useDateRange'

const props = defineProps<{
  accountId?: number
}>()

const { start, end } = useDateRange()
const chartRef = ref<HTMLDivElement>()
let chart: echarts.ECharts

async function fetchData() {
  const { data } = await chartApi.balance({
    start: start.value,
    end: end.value,
    account_id: props.accountId,
  })

  chart.setOption({
    tooltip: {
      trigger: 'axis',
      formatter: (params: any) => {
        const date = params[0].axisValue
        const values = params.map((p: any) => `${p.seriesName}: ¥${p.data}`).join('<br/>')
        return `${date}<br/>${values}`
      },
    },
    legend: { data: data.datasets.map(d => d.label) },
    xAxis: {
      type: 'category',
      data: data.labels,
    },
    yAxis: {
      type: 'value',
      axisLabel: {
        formatter: (val: number) => `¥${val.toLocaleString()}`,
      },
    },
    series: data.datasets.map(d => ({
      name: d.label,
      type: 'line',
      data: d.data,
      smooth: true,
    })),
  })
}

onMounted(() => {
  chart = echarts.init(chartRef.value!)
  fetchData()
  window.addEventListener('resize', () => chart.resize())
})

watch([start, end], () => fetchData())
</script>

<template>
  <div ref="chartRef" class="h-80 w-full" />
</template>
```

### 预算概览图 (BudgetOverviewChart)

```vue
<!-- components/chart/BudgetOverviewChart.vue -->
<script setup lang="ts">
import { ref, onMounted } from 'vue'
import * as echarts from 'echarts'
import { chartApi } from '@/api/charts'

const props = defineProps<{
  budgets: Budget[]
}>()

const chartRef = ref<HTMLDivElement>()
let chart: echarts.ECharts

onMounted(() => {
  chart = echarts.init(chartRef.value!)

  const data = props.budgets.map(b => ({
    name: b.name,
    value: Number(b.current_spent || 0),
    limit: Number(b.current_limit?.amount || 0),
  }))

  chart.setOption({
    tooltip: {
      trigger: 'item',
      formatter: '{b}: ¥{c}',
    },
    series: [{
      type: 'pie',
      radius: ['40%', '70%'],
      data,
      label: {
        formatter: '{b}\n¥{c}',
      },
    }],
  })
})
</script>

<template>
  <div ref="chartRef" class="h-64 w-full" />
</template>
```

## 6. 交易列表组件 (TransactionList)

```vue
<!-- components/transaction/TransactionList.vue -->
<script setup lang="ts">
import { onMounted } from 'vue'
import { usePagination } from '@/composables/usePagination'
import { transactionApi } from '@/api/transactions'
import { useCurrencyStore } from '@/stores/currency'
import { useDateRange } from '@/composables/useDateRange'
import { useRouter } from 'vue-router'

const router = useRouter()
const currencyStore = useCurrencyStore()
const { start, end } = useDateRange()

const {
  items: transactions,
  loading,
  page,
  total,
  totalPages,
  fetch,
} = usePagination<TransactionGroup>((params) =>
  transactionApi.list({ ...params, start: start.value, end: end.value })
)

function getTypeIcon(type: TransactionType) {
  switch (type) {
    case 'withdrawal': return 'arrow-up'  // 支出
    case 'deposit': return 'arrow-down'    // 收入
    case 'transfer': return 'swap'         // 转账
    default: return 'circle'
  }
}

function getTypeColor(type: TransactionType) {
  switch (type) {
    case 'withdrawal': return 'text-red-500'
    case 'deposit': return 'text-green-500'
    case 'transfer': return 'text-blue-500'
    default: return 'text-gray-500'
  }
}

onMounted(() => fetch())
</script>

<template>
  <div v-loading="loading">
    <div v-for="group in transactions" :key="group.id" class="transaction-group">
      <div
        v-for="journal in group.journals"
        :key="journal.id"
        class="transaction-item cursor-pointer hover:bg-gray-50 p-3 border-b"
        @click="router.push({ name: 'TransactionDetail', params: { id: group.id } })"
      >
        <div class="flex items-center justify-between">
          <div class="flex items-center gap-3">
            <el-icon :class="getTypeColor(journal.transaction_type_id)" size="20">
              <component :is="getTypeIcon(journal.transaction_type_id)" />
            </el-icon>
            <div>
              <div class="font-medium">{{ journal.description }}</div>
              <div class="text-sm text-gray-500">
                {{ journal.transactions[0]?.account?.name }}
                <span v-if="journal.transactions[1]"> → {{ journal.transactions[1]?.account?.name }}</span>
              </div>
            </div>
          </div>
          <div class="text-right">
            <div :class="getTypeColor(journal.transaction_type_id)" class="font-mono font-medium">
              {{ journal.transaction_type_id === 'withdrawal' ? '-' : '+' }}
              {{ currencyStore.formatAmount(journal.transactions[0]?.amount, journal.currency_id) }}
            </div>
            <div class="text-xs text-gray-400">{{ journal.date }}</div>
          </div>
        </div>
      </div>
    </div>

    <!-- 分页 -->
    <el-pagination
      v-model:current-page="page"
      :total="total"
      :page-size="50"
      layout="prev, pager, next"
      class="mt-4 justify-center"
      @current-change="fetch()"
    />
  </div>
</template>
```
