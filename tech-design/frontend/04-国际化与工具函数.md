## 2. 格式化工具

### 金额格式化

```typescript
// utils/format.ts
import Decimal from 'decimal.js'
import dayjs from 'dayjs'

/**
 * 格式化金额
 * @param amount 金额字符串
 * @param symbol 货币符号
 * @param decimalPlaces 小数位数
 * @param showSign 是否显示正负号
 */
export function formatAmount(
  amount: string | number,
  symbol: string = '¥',
  decimalPlaces: number = 2,
  showSign: boolean = false,
): string {
  const d = new Decimal(amount)
  const formatted = d.abs().toFixed(decimalPlaces).replace(/\B(?=(\d{3})+(?!\d))/g, ',')
  const sign = d.isNeg() ? '-' : (showSign && d.isPos() ? '+' : '')
  return `${sign}${symbol}${formatted}`
}

/**
 * 格式化百分比
 */
export function formatPercent(value: number, total: number): string {
  if (total === 0) return '0%'
  return `${new Decimal(value).div(total).mul(100).toFixed(1)}%`
}

/**
 * 格式化日期
 */
export function formatDate(date: string, format: string = 'YYYY-MM-DD'): string {
  return dayjs(date).format(format)
}

/**
 * 格式化相对时间
 */
export function formatRelativeTime(date: string): string {
  const diff = dayjs().diff(dayjs(date), 'day')
  if (diff === 0) return '今天'
  if (diff === 1) return '昨天'
  if (diff < 7) return `${diff}天前`
  if (diff < 30) return `${Math.floor(diff / 7)}周前`
  if (diff < 365) return `${Math.floor(diff / 30)}月前`
  return `${Math.floor(diff / 365)}年前`
}
```

```typescript
/**
 * 构建搜索查询字符串
 */
export function buildSearchQuery(operators: Array<{ key: string; value: string }>): string {
  return operators
    .filter(op => op.value)
    .map(op => {
      // 包含空格的值用引号包裹
      if (op.value.includes(' ')) {
        return `${op.key}:"${op.value}"`
      }
      return `${op.key}:${op.value}`
    })
    .join(' ')
}
```

## 5. 币种计算工具

```typescript
// utils/currency.ts
import Decimal from 'decimal.js'

/**
 * 将外币金额转换为本币
 */
export function convertToNative(
  amount: string,
  rate: string,
): string {
  return new Decimal(amount).mul(rate).toFixed(2)
}

/**
 * 计算账户余额
 * 资产账户: 期初余额 + 贷方总额 - 借方总额
 * 负债账户: 期初余额 + 借方总额 - 贷方总额
 */
export function calculateBalance(
  openingBalance: string,
  debitTotal: string,
  creditTotal: string,
  accountType: string,
): string {
  const opening = new Decimal(openingBalance || '0')
  const debit = new Decimal(debitTotal)
  const credit = new Decimal(creditTotal)

  if (['debt', 'loan', 'mortgage'].includes(accountType)) {
    return opening.plus(debit).minus(credit).toFixed(2)
  }

  return opening.plus(credit).minus(debit).toFixed(2)
}

/**
 * 计算储蓄罐进度百分比
 */
export function calculatePiggyBankProgress(
  currentAmount: string,
  targetAmount: string,
): number {
  if (new Decimal(targetAmount).isZero()) return 0
  const progress = new Decimal(currentAmount).div(targetAmount).mul(100).toNumber()
  return Math.min(progress, 100)
}
```
